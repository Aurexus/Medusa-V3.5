#!/bin/bash

# Install the Microsoft ODBC Driver 17 for SQL Server
curl https://packages.microsoft.com/keys/microsoft.asc | apt-key add -
curl https://packages.microsoft.com/config/debian/10/prod.list > /etc/apt/sources.list.d/mssql-release.list
apt-get update
ACCEPT_EULA=Y apt-get install -y msodbcsql17 

# 1️⃣ Activate virtual environment
source /home/gerard/env/bin/activate

# 2️⃣ Go to the folder containing app.py and imageapp/
cd /home/gerard/Medusa-V3.5   # <- replace <project-root> with actual folder

# 3️⃣ Set PYTHONPATH so Python can find imageapp
export PYTHONPATH=$(pwd)

# 4️⃣ Install dependencies if not already installed
pip install --user -r requirements.txt

# 5️⃣ Set Flask environment variables
export FLASK_ENV=production
export FLASK_APP=app.py

# 6️⃣ Start Flask with Gunicorn
gunicorn app:app --bind=0.0.0.0:8000 --timeout 300
